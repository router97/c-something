This is another problem that tests your ability to analyze the cache behavior of C
code. Assume we execute the three summation functions in Figure 6.47 under the
following conditions:

. sizeof(int) = 4.

. The machine has a 4 KB direct-mapped cache with a 16-byte block size.

. Within the two loops, the code uses memory accesses only for the array data.
The loop indices and the value sum are held in registers.

. Array a is stored starting at memory address 0x08000000.

Fill in the table for the approximate cache miss rate for the two cases N = 64
and N = 60.

| Function | N = 64 | N = 60 |
|----------|--------|--------|
| sumA     | 25%    | 25%    |
| sumB     | 100%   | 25%    |
| sumC     | 25%    | 25%    |

typedef int array_t[N][N];

int sumA(array_t a)
{
    int i, j;
    int sum = 0;
    for (i = 0; i < N; i++)
        for (j = 0; j < N; j++) {
            sum += a[i][j];
        }
    return sum;
}
/*
stride-4(bytes) pattern
accesses memory blocks sequentially, so one miss every 
(number of elements that fit in a block) reads

1 / 4 * 100 = 25%

changing N won't affect it
*/


int sumB(array_t a)
{
    int i, j;
    int sum = 0;
    for (j = 0; j < N; j++)
        for (i = 0; i < N; i++) {
            sum += a[i][j];
        }
    return sum;
}
/*
stride-256(bytes) pattern

4096 / 16 = 256 (lines in cache)
64 * 64 * 4 = 16384 bytes (size of array)
16384 / 16 = 1024 blocks (size of array)
256 / 16 = 16 blocks (stride)

OP          Elem            Status
load        a[0][0]         MISS :: load a[0][0-3] into line 0
load        a[1][0]         MISS :: load a[1][0-3] into line 16
...
load        a[15][0]        MISS :: load a[15][0-3] into line 240
load        a[16][0]        MISS :: load a[16][0-3] into line 0
...
load        a[48][0]        MISS :: load a[48][0-3] into line 0
...
load        a[63][0]        MISS :: load a[63][0-3] into line 240
...

the same is going to happen with all other iterations

miss rate is 100% is n=64

if n=60

60 * 60 * 4 = 14400 bytes (size of array)
14400 / 16 = 900 blocks (size of array)
60 * 4 = 240 bytes (stride)
240 / 16 = 15 blocks (stride)

OP          Elem            Status
load        a[0][0]         MISS :: load a[0][0-3] into line 0
load        a[1][0]         MISS :: load a[1][0-3] into line 15
...
load        a[17][0]        MISS :: load a[17][0-3] into line 255
load        a[18][0]        MISS :: load a[18][0-3] into line 14
...
load        a[34][0]        MISS :: load a[34][0-3] into line 254
load        a[35][0]        MISS :: load a[35][0-3] into line 13
...
load        a[51][0]        MISS :: load a[51][0-3] into line 253
load        a[52][0]        MISS :: load a[52][0-3] into line 12
...
load        a[59][0]        MISS :: load a[59][0-3] into line 117
...
load        a[0][1]         HIT
load        a[1][1]         HIT
...
load        a[59][1]        HIT
...
load        a[59][3]        HIT
...
load        a[0][4]         MISS :: load a[0][4-7] into line 1
...
load        a[0][5]         HIT

First row loading will always miss, then following 3 will hit

miss rate 25%
*/


int sumC(array_t a)
{
    int i, j;
    int sum = 0;
    for (j = 0; j < N; j+=2)
        for (i = 0; i < N; i+=2) {
            sum += (a[i][j] + a[i+1][j]
                    + a[i][j+1] + a[i+1][j+1]);
        }
    return sum;
}
/*
for n = 64, 25%

for n = 60, 25%
*/
