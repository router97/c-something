/*
Transposing the rows and columns of a matrix is an important problem in signal
processing and scientific computing applications. It is also interesting from a 
locality point of view because its reference pattern is both row-wise and column-wise.
For example, consider the following transpose routine:
*/

typedef int array[2][2];

void transpose1(array dst, array src)
{
    int i, j;

    for (i = 0; i < 2; i++) {
        for (j = 0; j < 2; j++) {
            dst[j][i] = src[i][j];
        }
    }
}
